## 二、API 调用层开发

### 2.1 创建 API 函数文件

- [ ] 创建 `api/emr-status.js` 文件
  - 路径: `src/api/emr-status.js`
  
  函数定义:
  ```javascript
  import request from '@/utils/request'
  
  /**
   * 查询患者病历状态（包含门诊和急诊）
   * @param {Object} params - 查询参数
   * @param {string} params.patientId - 患者ID（必填）
   * @param {string} params.encounterId - 就诊ID（可选）
   * @param {string} params.emrType - 病历类型 OPT/EMG/ALL（可选）
   * @param {number} params.pageNum - 页码
   * @param {number} params.pageSize - 每页条数
   * @returns {Promise}
   */
  export function queryPatientEmrStatus(params) {
    return request({
      url: '/api/v1/opt/emr/emg-status/patient/status',
      method: 'get',
      params
    })
  }
  ```

## 三、组件开发

### 3.1 创建病历类型标签组件

- [ ] 创建 `EmrTypeTag.vue` 组件
  - 路径: `src/components/common/EmrTypeTag.vue`
  
  组件功能:
  ```vue
  <template>
    <el-tag 
      :type="getTagType(emrType)" 
      :size="size"
      effect="plain"
    >
      {{ getEmrTypeName(emrType) }}
    </el-tag>
  </template>
  
  <script>
  import { EmrTypeEnum, EmrTypeNameMap } from '@/types/emr-status'
  
  export default {
    name: 'EmrTypeTag',
    props: {
      emrType: {
        type: String,
        required: true
      },
      size: {
        type: String,
        default: 'mini'
      }
    },
    methods: {
      getEmrTypeName(type) {
        return EmrTypeNameMap[type] || type
      },
      getTagType(type) {
        const typeMap = {
          [EmrTypeEnum.OPT]: '',           // 门诊 - 默认灰色
          [EmrTypeEnum.EMG]: 'danger',      // 急诊 - 红色
          [EmrTypeEnum.INP]: 'warning'      // 住院 - 橙色
        }
        return typeMap[type] || 'info'
      }
    }
  }
  </script>
  ```

     - 调用新的 API `queryPatientEmrStatus`

  主要代码变更:
  ```vue
  <template>
    <div class="record-current">
      <!-- 筛选区域 -->
      <el-form :inline="true" :model="queryForm">
        <el-form-item label="病历类型">
          <el-select v-model="queryForm.emrType" @change="handleQuery">
            <el-option label="全部" value=""></el-option>
            <el-option label="门诊" value="OPT"></el-option>
            <el-option label="急诊" value="EMG"></el-option>
          </el-select>
        </el-form-item>
      </el-form>
      
      <!-- 列表区域 -->
      <el-table :data="recordList">
        <el-table-column label="类型" width="80">
          <template slot-scope="scope">
            <emr-type-tag :emr-type="scope.row.emrType" />
          </template>
        </el-table-column>
        <el-table-column label="病历名称" prop="emrTemplateName"></el-table-column>
        <el-table-column label="状态" width="100">
          <template slot-scope="scope">
            <emr-status-tag :status="scope.row.status" />
          </template>
        </el-table-column>
        <!-- 其他列 -->
      </el-table>
    </div>
  </template>
  
  <script>
  import { queryPatientEmrStatus } from '@/api/emr-status'
  import EmrTypeTag from '@/components/common/EmrTypeTag'
  import EmrStatusTag from '@/components/common/EmrStatusTag'
  
  export default {
    components: {
      EmrTypeTag,
      EmrStatusTag
    },
    data() {
      return {
        queryForm: {
          patientId: '',
          encounterId: '',
          emrType: '' // OPT/EMG/ALL
        },
        recordList: []
      }
    },
    methods: {
      async handleQuery() {
        const params = {
          ...this.queryForm,
          pageNum: 1,
          pageSize: 100
        }
        const res = await queryPatientEmrStatus(params)
        if (res.code === 0) {
          this.recordList = res.data.list || []
        }
      }
    }
  }
  </script>
  ```
